    <script>
    (function() {
      'use strict';
      var filterBtns = document.querySelectorAll('.gallery-filter-btn');
      var items = document.querySelectorAll('.gallery-item');
      var popup = document.getElementById('mv-gallery-popup');
      var popupImg = document.getElementById('mv-popup-img');
      var closeBtn = document.getElementById('mv-popup-close');
      var nextBtn = document.getElementById('mv-popup-next');
      var prevBtn = document.getElementById('mv-popup-prev');
      var galleryImages = [];
      var currentIndex = 0;

      if (!filterBtns.length || !items.length) return;

      // Build image list from currently visible items
      function buildImageList() {
        galleryImages = [];
        items.forEach(function(item) {
          if (item.style.display !== 'none') {
            var img = item.querySelector('img');
            if (img) galleryImages.push(img.src);
          }
        });
      }

      // Filter handler
      filterBtns.forEach(function(btn) {
        btn.addEventListener('click', function() {
          filterBtns.forEach(function(b) { b.classList.remove('active'); });
          btn.classList.add('active');
          var filter = btn.getAttribute('data-filter');
          var grid = document.querySelector('.mv-gallery-grid');
          if (grid) {
              if (filter === 'all') {
                  grid.classList.remove('is-filtered');
              } else {
                  grid.classList.add('is-filtered');
              }
          }
          items.forEach(function(item) {
            if (filter === 'all' || item.getAttribute('data-category') === filter) {
              item.style.display = 'block';
            } else {
              item.style.display = 'none';
            }
          });
          buildImageList();
        });
      });

      // Initial image list
      buildImageList();

      // Lightbox open
      items.forEach(function(item) {
        var card = item.querySelector('.mv-gallery-card');
        if (!card) return;
        card.addEventListener('click', function(e) {
          e.preventDefault();
          var img = item.querySelector('img');
          if (!img || !popup) return;
          buildImageList();
          currentIndex = galleryImages.indexOf(img.src);
          if (currentIndex < 0) currentIndex = 0;
          popupImg.src = img.src;
          popupImg.alt = img.alt || '';
          popup.classList.add('active');
          document.body.style.overflow = 'hidden';
        });
      });

      // Close popup
      if (closeBtn) {
        closeBtn.addEventListener('click', function() {
          popup.classList.remove('active');
          document.body.style.overflow = '';
        });
      }
      if (popup) {
        popup.addEventListener('click', function(e) {
          if (e.target === popup) {
            popup.classList.remove('active');
            document.body.style.overflow = '';
          }
        });
      }
      document.addEventListener('keydown', function(e) {
        if (!popup || !popup.classList.contains('active')) return;
        if (e.key === 'Escape') {
          popup.classList.remove('active');
          document.body.style.overflow = '';
        }
        if (e.key === 'ArrowRight') {
          currentIndex = (currentIndex + 1) % galleryImages.length;
          popupImg.src = galleryImages[currentIndex];
        }
        if (e.key === 'ArrowLeft') {
          currentIndex = (currentIndex - 1 + galleryImages.length) % galleryImages.length;
          popupImg.src = galleryImages[currentIndex];
        }
      });

      // Next / Prev
      if (nextBtn) {
        nextBtn.addEventListener('click', function() {
          currentIndex = (currentIndex + 1) % galleryImages.length;
          popupImg.src = galleryImages[currentIndex];
        });
      }
      if (prevBtn) {
        prevBtn.addEventListener('click', function() {
          currentIndex = (currentIndex - 1 + galleryImages.length) % galleryImages.length;
          popupImg.src = galleryImages[currentIndex];
        });
      }
    })();
    </script>
